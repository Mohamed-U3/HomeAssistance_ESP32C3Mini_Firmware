84static void webSocketEvent(uint8_t num, WStype_t type, uint8_t *payload, size_t length)
85{
86 if (type == WStype_CONNECTED)
87 {
88 String json = buildStateJson();
89 webSocket.sendTXT(num, json);
90 }
91 else if (type == WStype_TEXT)
92 {
93 // Build bounded string using length (WebSockets payload is not guaranteed null-terminated)
94 String msg;
95 msg.reserve(length);
96 for (size_t i = 0; i < length; i++)
97 msg += static_cast<char>(payload[i]);
98
99 if (msg.startsWith("TOGGLE:"))
100 {
101 String pinStr = msg.substring(7);
102 if (isPositiveInteger(pinStr))
103 toggleSingleRelayByPin(pinStr.toInt());
104 }
105 else if (msg == "ALL_ON")
106 {
107 setAllRelays(true);
108 }
109 else if (msg == "ALL_OFF")
110 {
111 setAllRelays(false);
112 }
113 else if (msg == "TOGGLE_ALL")
114 {
115 toggleAllRelays();
116 }
117 else if (msg == "SMART_TOGGLE")
118 {
119 smartToggleAllRelays();
120 }
121 else if (msg.startsWith("SET_POWER_MODE:"))
122 {
123 String modeStr = msg.substring(15);
124 if (isPositiveInteger(modeStr))
125 {
126 int mode = modeStr.toInt();
127 if (mode >= POWER_ON_ALL_OFF && mode <= POWER_ON_RESTORE)
128 {
129 savePowerOnMode(static_cast<uint8_t>(mode));
130 if (appConfig.power_on_mode == POWER_ON_RESTORE)
131 persistCurrentRelayMask();
132 broadcastStates();
133 }
134 }
135 }
136 }

458void handleWebUi()
459{
460 server.handleClient();
461 webSocket.loop();
462
463 unsigned long now = millis();
464 if (pendingPortal && now >= pendingActionAt)
465 {
466 pendingPortal = false;
467 if (portalActionCb)
468 portalActionCb();
469 }
470 if (pendingReboot && now >= pendingActionAt)
471 {
472 pendingReboot = false;
473 publishMqttOffline();
474 delay(100);
475 ESP.restart();
476 }
477
478 // Push periodic updates to WebSocket clients when MQTT or RSSI changes noticeably
479 if (now - lastStatusBroadcast >= 5000)
480 {
481 lastStatusBroadcast = now;
482 bool mqttNow = isMqttConnected();
483 int rssiNow = WiFi.RSSI();
484 if (mqttNow != lastKnownMqtt || abs(rssiNow - lastKnownRssi) >= 4)
485 {
486 broadcastStates();
487 }
488 }
