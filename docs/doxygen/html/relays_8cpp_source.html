155void handleButtons(PortalTriggerCallback onPortalRequested)
156{
157 unsigned long now = millis();
158
159 // Master toggle button (short press -> smart toggle all)
160 int masterReading = digitalRead(MASTER_BTN_PIN);
161 if (masterReading != lastMasterReading)
162 lastMasterDebounce = now;
163
164 if ((now - lastMasterDebounce) > DEBOUNCE_DELAY_MS && masterReading != masterStableState)
165 {
166 masterStableState = masterReading;
167 if (masterStableState == LOW)
168 smartToggleAllRelays();
169 }
170 lastMasterReading = masterReading;
171
172 // Config portal button (3-second hold to prevent accidental triggers)
173 int configReading = digitalRead(CONFIG_BTN_PIN);
174 if (configReading != lastConfigReading)
175 lastConfigDebounce = now;
176
177 if ((now - lastConfigDebounce) > DEBOUNCE_DELAY_MS && configReading != configStableState)
178 {
179 configStableState = configReading;
180 if (configStableState == LOW)
181 {
182 configPressStart = now;
183 configTriggered = false;
184 }
185 else
186 {
187 configTriggered = false;
188 }
189 }
190 lastConfigReading = configReading;
191
192 if (configStableState == LOW && !configTriggered && (now - configPressStart) >= CONFIG_HOLD_MS)
193 {
194 configTriggered = true;
195 if (onPortalRequested)
196 onPortalRequested();
197 }
