26void sanitizeHostname(char *buf, size_t bufSize)
27{
28 if (!buf || bufSize == 0)
29 return;
30
31 char clean[33];
32 size_t out = 0;
33 size_t maxCopy = (bufSize - 1 < sizeof(clean) - 1) ? (bufSize - 1) : (sizeof(clean) - 1);
34
35 for (size_t i = 0; buf[i] != '\0' && out < maxCopy; i++)
36 {
37 unsigned char c = static_cast<unsigned char>(buf[i]);
38 if (isalnum(c) || c == '-')
39 {
40 clean[out++] = static_cast<char>(c);
41 }
42 else if (c == '_' || c == ' ')
43 {
44 clean[out++] = '-';
45 }
46 }
47 clean[out] = '\0';
48
49 // Trim leading/trailing hyphens for valid mDNS hostnames
50 size_t start = 0;
51 while (clean[start] == '-')
52 start++;
53
54 size_t end = strlen(clean);
55 while (end > start && clean[end - 1] == '-')
56 end--;
57
58 if (end <= start)
59 {
60 strlcpy(buf, DEFAULT_HOSTNAME, bufSize);
61 return;
62 }
63
64 size_t len = end - start;
65 if (len >= bufSize)
66 len = bufSize - 1;
67 memmove(buf, clean + start, len);
68 buf[len] = '\0';
69}

185{

208bool connectSavedWifi(unsigned long timeoutMs, WifiTickCallback onTick)
209{
210 if (!hasSavedWifi())
211 return false;
212
213 WiFi.mode(WIFI_STA);
214 WiFi.setHostname(appConfig.hostname);
215 WiFi.setAutoReconnect(true);
216 WiFi.begin();
217
218 unsigned long start = millis();
219 unsigned long lastRetry = start;
220
221 while (millis() - start < timeoutMs)
222 {
223 if (WiFi.status() == WL_CONNECTED)
224 return true;
225
226 unsigned long now = millis();
227 if (now - lastRetry >= WIFI_RETRY_INTERVAL_MS)
228 {
229 lastRetry = now;
230 WiFi.disconnect(false, false);
231 delay(50);
232 WiFi.begin();
233 }
234
235 if (onTick && onTick(true))
236 return false;
237
238 delay(10);
239 }
240
241 return WiFi.status() == WL_CONNECTED;
242}
