89static void mqttCallback(char *topic, byte *payload, unsigned int length)
90{
91 String prefix = String("homeassistant/switch/") + appConfig.hostname + "/relay";
92 String strTopic(topic);
93 if (!strTopic.startsWith(prefix))
94 return;
95
96 String rest = strTopic.substring(prefix.length());
97 int slashIdx = rest.indexOf('/');
98 if (slashIdx <= 0 || rest.substring(slashIdx) != "/set")
99 return;
100
101 String idxStr = rest.substring(0, slashIdx);
102 for (unsigned int i = 0; i < idxStr.length(); i++)
103 {
104 if (!isdigit(static_cast<unsigned char>(idxStr[i])))
105 return;
106 }
107
108 int index = idxStr.toInt();
109 if (index < 0 || static_cast<size_t>(index) >= RELAY_COUNT)
110 return;
111
112 if (length == 2 && memcmp(payload, "ON", 2) == 0)
113 {
114 setRelay(static_cast<size_t>(index), true);
115 }
116 else if (length == 3 && memcmp(payload, "OFF", 3) == 0)
117 {
118 setRelay(static_cast<size_t>(index), false);
119 }
120}

122static void mqttProbeTask(void *pvParameters)
123{
124 (void)pvParameters;
125
126 for (;;)
127 {
128 bool shouldProbe = false;
129 if (!mqttPaused && WiFi.status() == WL_CONNECTED && !mqttClient.connected() && strlen(appConfig.mqtt_server) > 0)
130 {
131 if (probeMutex && xSemaphoreTake(probeMutex, pdMS_TO_TICKS(50)) == pdTRUE)
132 {
133 shouldProbe = !mqttBrokerReachable;
134 xSemaphoreGive(probeMutex);
135 }
136 }
137
138 if (shouldProbe)
139 {
140 IPAddress resolved;
141 int port = atoi(appConfig.mqtt_port);
142 if (port <= 0 || port > 65535)
143 port = 1883;
144
145 if (WiFi.hostByName(appConfig.mqtt_server, resolved))
146 {
147 WiFiClient probeClient;
148 if (probeClient.connect(resolved, port, 1500))
149 {
150 probeClient.stop();
151 if (probeMutex && xSemaphoreTake(probeMutex, pdMS_TO_TICKS(100)) == pdTRUE)
152 {
153 mqttResolvedIP = resolved;
154 mqttBrokerReachable = true;
155 xSemaphoreGive(probeMutex);
156 }
157 }
158 }
159 }
160
161 vTaskDelay(pdMS_TO_TICKS(5000));
162 }

165static void reconnectMQTT()
166{
167 if (mqttPaused || mqttClient.connected())
168 return;
169
170 unsigned long now = millis();
171 if (now - lastMqttReconnectAttempt <= 5000)
172 return;
173
174 bool reachable = false;
175 IPAddress targetIP;
176
177 if (probeMutex && xSemaphoreTake(probeMutex, pdMS_TO_TICKS(20)) == pdTRUE)
178 {
179 if (mqttBrokerReachable)
180 {
181 reachable = true;
182 targetIP = mqttResolvedIP;
183 mqttBrokerReachable = false; // Require a fresh probe if this attempt fails
184 }
185 xSemaphoreGive(probeMutex);
186 }
187
188 if (!reachable)
189 return;
190
191 lastMqttReconnectAttempt = now;
192
193 int port = atoi(appConfig.mqtt_port);
194 if (port <= 0 || port > 65535)
195 port = 1883;
196
197 mqttClient.setServer(targetIP, port);
198
199 Serial.printf("Connecting to MQTT (%s / %s:%d)...",
200 appConfig.mqtt_server, targetIP.toString().c_str(), port);
201
202 String clientId = String(appConfig.hostname) + "-" + String(random(0xffff), HEX);
203 String availTopic = getAvailabilityTopic();
204 bool connected = false;
205
206 if (strlen(appConfig.mqtt_user) > 0)
207 {
208 connected = mqttClient.connect(
209 clientId.c_str(),
210 appConfig.mqtt_user,
211 appConfig.mqtt_pass,
212 availTopic.c_str(),
213 1,
214 true,
215 "offline");
216 }
217 else
218 {
219 connected = mqttClient.connect(
220 clientId.c_str(),
221 availTopic.c_str(),
222 1,
223 true,
224 "offline");
225 }
226
227 if (connected)
228 {
229 Serial.println(" Connected!");
230 mqttClient.publish(availTopic.c_str(), "online", true);
231
232 for (size_t i = 0; i < RELAY_COUNT; i++)
233 {
234 String cmdTopic = getRelayTopic(i, "set");
235 mqttClient.subscribe(cmdTopic.c_str());
236 }
237
238 sendHADiscovery();
239 }
240 else
241 {
242 Serial.printf(" Failed, rc=%d\n", mqttClient.state());
243 }
