106void loop()
107{

112
113 // Monitor runtime Wi-Fi connectivity: retry for 2 minutes, then open Config Portal AP
114 if (WiFi.status() == WL_CONNECTED)
115 {
116 wifiDisconnectedSince = 0;
117 }
118 else
119 {
120 unsigned long now = millis();
121 if (wifiDisconnectedSince == 0)
122 {
123 wifiDisconnectedSince = now;
124 lastWifiRetry = now;
125 Serial.println("[WiFi] Connection lost. Searching for saved SSID for 2 minutes...");
126 }
127 else if (now - lastWifiRetry >= WIFI_RETRY_INTERVAL_MS)
128 {
129 lastWifiRetry = now;
130 WiFi.reconnect();
131 }
132
133 if (now - wifiDisconnectedSince >= WIFI_SEARCH_TIMEOUT_MS)
134 {
135 Serial.println("[WiFi] Saved SSID unreachable for 2 minutes. Opening Config Portal AP...");
136 openConfigPortal();
137 }
138 }
